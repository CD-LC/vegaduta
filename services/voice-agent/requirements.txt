# voice-agent: real-time LiveKit voice pipeline worker.
#
# Versions below are pinned to what was actually pip-installed into a
# scratch venv and verified against (see README.md's "What was verified"
# section) - not guessed from memory. Re-verify against
# https://github.com/livekit/agents releases before bumping; this SDK's
# public API (AgentServer/rtc_session vs. the older WorkerOptions/cli.run_app
# shape both this file and main.py rely on) has moved fast across 1.x.
livekit-agents==1.8.8

# livekit-agents 1.8.8 requires openai>=3.19,<4; the 1.6.8-era 2.x hold
# (its inference/llm.py passed httpx objects to openai) is gone.
openai==3.35.1

# Official ElevenLabs TTS plugin - this platform's preferred/default TTS
# provider (see core/agents/.../SpeakToolCallback.java's javadoc: "ElevenLabs
# first... Sarvam otherwise"). Only the TTS leg is used from this package;
# its `stt` submodule (ElevenLabs also offers STT) is NOT used here - STT is
# sidecar Whisper (app/stt_sidecar_whisper.py), per the build brief.
livekit-plugins-elevenlabs==1.8.8

# Silero VAD - turn detection (utterance start/end), ships its own ONNX
# model file inside the wheel (see silero/resources/silero_vad.onnx) so,
# unlike sidecar's sentence-transformers model, there is nothing to
# download or cache at runtime/build time - no HF_HOME-style baking needed
# in the Dockerfile for this dependency.
livekit-plugins-silero==1.8.8

# HTTP client for both custom plugins (stt_sidecar_whisper.py's sidecar
# /transcribe call, llm_core_bridge.py's core /api/agents/{id}/chat SSE
# call). Already a transitive dependency of livekit-agents itself; pinned
# directly here since both of this package's own modules import it by name.
aiohttp==3.14.5

# Perception ("sight") mode frame handling - see perception.py. numpy backs
# the scene-change gate (mean-abs-diff on a 64x36 grayscale thumbnail per
# decoded frame - must be vectorized, it runs at camera fps); Pillow does
# the rtc.VideoFrame RGB24 buffer -> resized/grayscale thumbnail and the
# JPEG encode (max 1024px long edge, q80) for the frames actually sent to
# core. numpy is already a transitive dependency of livekit-agents (its
# rtc audio/video frame plumbing); pinned directly here since perception.py
# imports it by name. Both pins verified present on PyPI for python:3.12
# (this image's base) but NOT pip-installed/runtime-verified in this pass -
# see the header comment's re-verify note.
numpy==2.5.4
pillow==13.0.0

# Local TTS provider - app/tts_piper.py's subprocess `piper` binary comes
# from this package's console entry point (piper-tts ships the `piper` CLI
# plus its onnxruntime dependency; the voice .onnx itself is baked into the
# image by the Dockerfile's download_voices step, not fetched at runtime).
piper-tts==1.8.0
